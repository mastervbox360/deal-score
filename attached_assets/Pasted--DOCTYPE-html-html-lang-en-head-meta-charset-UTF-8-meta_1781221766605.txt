  var icon = document.getElementById('lb-hide-icon');
  var label = document.getElementById('lb-hide-label');

function notesRelTime(ts){
  var diff = Date.now() - ts;
  var mins = Math.round(diff/60000);
  if(mins < 1) return 'just now';
  if(mins < 60) return mins + (mins===1 ? ' minute ago' : ' minutes ago');
  var hrs = Math.round(mins/60);
  if(hrs < 24) return hrs + (hrs===1 ? ' hour ago' : ' hours ago');
  var days = Math.round(hrs/24);
  return days + (days===1 ? ' day ago' : ' days ago');
}

  var list = document.getElementById('notesList');
  var count = document.getElementById('notesCount');
  count.textContent = notesData.length + (notesData.length===1 ? ' note' : ' notes');
  if(!notesData.length){
    list.innerHTML = '<div class="notes-empty"><i class="ti ti-notes"></i>No notes yet — add the first one above.</div>';
    return;
  }
  list.innerHTML = notesData.map(function(n,i){
    return '<div class="notes-item">'
      + '<div class="notes-item-hdr"><div class="notes-author"><i class="ti ti-user-circle" style="font-size:14px;"></i> ' + n.author + '</div>'
      + '<div style="display:flex;align-items:center;gap:8px;"><span class="notes-time">' + notesRelTime(n.time) + '</span>'
      + '<button class="notes-del" onclick="deleteNote(' + i + ')" title="Delete note"><i class="ti ti-trash"></i></button></div></div>'
      + '<div class="notes-text">' + n.text.replace(/</g,'&lt;') + '</div>'
      + '</div>';
  }).join('');
}

  if(!name) return;
  var isCommitted = invpickSelectedType==='committed';
  var phone = document.getElementById('invpickPhone').value.trim();
  var email = document.getElementById('invpickEmail').value.trim();
  var contact = [phone, email].filter(Boolean).join(' · ');
  var listId = isCommitted ? 'committed-list' : 'openmkt-list';
  var countId = isCommitted ? 'committed-count' : 'openmkt-count';
  injectInvestorCard(listId, countId, name, 'Sent', isCommitted, contact);
  closeInvestorPicker();
  showToast(name + ' created and added to deal · saved to Investors CRM');
  /* reset form */
  document.getElementById('invpickName').value = '';
  document.getElementById('invpickPhone').value = '';
  document.getElementById('invpickEmail').value = '';
  invpickSelectType('committed');
  invpickToggleCreateBtn();
}

function dismissExp(btn){
  var card=btn.closest?btn.closest(".exp-card"):null;
  if(!card){card=btn.parentNode;while(card&&!card.classList.contains("exp-card"))card=card.parentNode;}
  if(!card)return;
  card.style.display="none";
  var r=card.nextElementSibling;
  if(r&&r.classList&&r.classList.contains("exp-restore"))r.classList.add("show");
}
function restoreExp(btn){
  btn.classList.remove("show");
  var card=btn.previousElementSibling;
  if(card&&card.classList&&card.classList.contains("exp-card"))card.style.display="flex";
}

  // Simulated deal-aware notifications

      deal:'14 Roath Court Rd', msg:'Seller details not yet completed — add vendor motivation before sharing packs', time:fmt(noteDate2) },
    { id:'n5', unread:false, type:'red',   icon:'ti-clock-exclamation',
      deal:'52 Crwys Rd', msg:'Deal has been in Negotiation for 18 days with no update', time:fmt(noteDate2) },
  ];
}());
function renderNotifs(){
  var unread = notifData.filter(function(n){ return n.unread; }).length;
  var badge = document.getElementById('notif-badge');
  if(badge){ badge.classList.toggle('show', unread > 0); }
  var list = document.getElementById('notif-list');
  if(!list) return;
  if(!notifData.length){
    list.innerHTML = '<div class="notif-empty"><i class="ti ti-bell-off"></i><div class="notif-empty-msg">All caught up — no new notifications</div></div>';
    return;
  }
  list.innerHTML = notifData.map(function(n){
    return '<div class="notif-item'+(n.unread?' unread':'')+'" id="ni-'+n.id+'" onclick="readNotif(\''+n.id+'\')" >'
      +'<div class="notif-icon-wrap '+n.type+'"><i class="ti '+n.icon+'"></i></div>'
      +'<div class="notif-body">'
      +'<div class="notif-deal-tag">'+n.deal+'</div>'
      +'<div class="notif-msg">'+n.msg+'</div>'
      +'<div class="notif-time">'+n.time+'</div>'
      +'</div>'
      +(n.unread ? '<div class="notif-unread-dot"></div>' : '')
      +'</div>';
  }).join('');
}
function toggleNotif(e){
  e.stopPropagation();
  notifOpen = !notifOpen;
  var drop = document.getElementById('notif-drop');
  if(drop) drop.classList.toggle('show', notifOpen);
  if(notifOpen) renderNotifs();
}
function readNotif(id){
  var n = notifData.find(function(x){ return x.id===id; });
  if(n) n.unread = false;
  renderNotifs();
}
function markAllRead(){
  notifData.forEach(function(n){ n.unread=false; });
  renderNotifs();
}
document.addEventListener('click', function(e){
  if(!e.target.closest('.notif-wrap')){ notifOpen=false; var d=document.getElementById('notif-drop'); if(d) d.classList.remove('show'); }
});

document.addEventListener("DOMContentLoaded",function(){ renderNotifs(); });

/* ══ INVESTOR RESPONSE LOG ══ */
var respLogInvestorId = '';
var respLogInvestorName = '';
var respLogSelected = 'interested';
var investorTimelines = {};

function openRespLog(investorId, name){
  respLogInvestorId = investorId;
  respLogInvestorName = name;
  respLogSelected = 'interested';
  document.getElementById('rlm-investor-name').textContent = name;
  document.getElementById('rlm-note').value = '';
  document.querySelectorAll('.rlm-chip').forEach(function(c){ c.classList.remove('on'); });
  var first = document.querySelector('.rlm-chip');
  if(first) first.classList.add('on');
  document.getElementById('resp-log-overlay').classList.add('show');
}
function closeRespLog(){
  document.getElementById('resp-log-overlay').classList.remove('show');
}
function selectRespChip(el){
  document.querySelectorAll('#rlm-chips .rlm-chip').forEach(function(c){ c.classList.remove('on'); });
  el.classList.add('on');
  respLogSelected = el.getAttribute('data-val');
}
function saveRespLog(){
  var note = document.getElementById('rlm-note').value.trim();
  var labels = { interested:'Expressed interest', viewing:'Viewing booked', offer:'Offer made', called:'Called', emailed:'Emailed', passed:'Passed' };
  var colors = { interested:'teal', viewing:'navy', offer:'amber', called:'navy', emailed:'navy', passed:'red' };
  var label = labels[respLogSelected] || respLogSelected;
  var color = colors[respLogSelected] || 'navy';
  if(!investorTimelines[respLogInvestorId]) investorTimelines[respLogInvestorId] = [];
  investorTimelines[respLogInvestorId].unshift({ label:label, note:note, time:'Just now', color:color });
  var tl = document.getElementById('tl-'+respLogInvestorId);
  var toggle = tl ? tl.previousElementSibling : null;
  if(tl){
    var newItem = '<div class="inv-tl-item '+color+'">'
      +'<div class="inv-tl-body"><div class="inv-tl-msg">'+label+'</div>'
      +(note ? '<div class="inv-tl-note">"'+note+'"</div>' : '')
      +'</div><div class="inv-tl-time">Just now</div></div>';
    tl.insertAdjacentHTML('afterbegin', newItem);
    tl.classList.add('show');
    if(toggle){
      var count = tl.querySelectorAll('.inv-tl-item').length;
      toggle.innerHTML = '<i class="ti ti-history"></i> '+count+' response'+(count!==1?'s':'')+' logged';
    }
  }
  closeRespLog();
  showToast('Response logged for '+respLogInvestorName);
}
function toggleTimeline(id, btn){
  var tl = document.getElementById('tl-'+id);
  if(!tl) return;
  var open = tl.classList.toggle('show');
  var icon = btn.querySelector('i');
  if(icon) icon.style.transform = open ? 'rotate(180deg)' : '';
}
