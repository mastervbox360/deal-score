function dismissExp(btn){
  var card=btn.closest?btn.closest(".exp-card"):null;
  if(!card){card=btn.parentNode;while(card&&!card.classList.contains("exp-card"))card=card.parentNode;}
  if(!card)return;
  card.style.display="none";
  var r=card.nextElementSibling;
  if(r&&r.classList&&r.classList.contains("exp-restore"))r.classList.add("show");
}

  btn.classList.remove("show");
  var card=btn.previousElementSibling;
  if(card&&card.classList&&card.classList.contains("exp-card"))card.style.display="flex";
}

function notesRelTime(ts){
  var diff = Date.now() - ts;
  var mins = Math.round(diff/60000);
  if(mins < 1) return 'just now';
  if(mins < 60) return mins + (mins===1 ? ' minute ago' : ' minutes ago');
  var hrs = Math.round(mins/60);
  if(hrs < 24) return hrs + (hrs===1 ? ' hour ago' : ' hours ago');
  var days = Math.round(hrs/24);
  return days + (days===1 ? ' day ago' : ' days ago');
}

    return;

/* ══ CLOSE DEAL ══ */
var closeDealOutcome = 'sold';
function openCloseDeal(){
  var d = new Date(); d.setDate(d.getDate());
  document.getElementById('cdm-date').value = d.toISOString().split('T')[0];
  document.getElementById('close-deal-overlay').classList.add('show');
  selectOutcome('sold');
}
function closeCloseDeal(){
  document.getElementById('close-deal-overlay').classList.remove('show');
}
function selectOutcome(outcome){
  closeDealOutcome = outcome;
  ['sold','exchanged','fell-through','withdrawn'].forEach(function(o){
    document.getElementById('oc-'+o).classList.remove('on');
  });
  document.getElementById('oc-'+outcome).classList.add('on');
  var fieldsEl = document.getElementById('cdm-fields');
  var showFields = outcome === 'sold' || outcome === 'exchanged';
  fieldsEl.classList.toggle('show', showFields);
  var btn = document.getElementById('cdm-confirm-btn');
  var isDanger = outcome === 'fell-through' || outcome === 'withdrawn';
  btn.classList.toggle('danger', isDanger);
  btn.textContent = isDanger ? 'Mark as withdrawn' : 'Complete deal';
}
function confirmCloseDeal(){
  closeCloseDeal();
  var isComplete = closeDealOutcome === 'sold' || closeDealOutcome === 'exchanged';
  var outcomeLabels = { sold:'Sold — deal complete', exchanged:'Exchanged — awaiting completion', 'fell-through':'Fell through — marked withdrawn', withdrawn:'Withdrawn — deal removed' };
  var msg = outcomeLabels[closeDealOutcome];
  var banner = document.getElementById('archived-banner');
  banner.classList.remove('complete');
  if(isComplete) banner.classList.add('complete');
  banner.classList.add('show');
  document.getElementById('archived-msg').textContent = msg;
  showToast(isComplete ? 'Deal marked as complete' : 'Deal marked as withdrawn');
}
function undoArchive(){
  document.getElementById('archived-banner').classList.remove('show');
  showToast('Deal restored to active pipeline');
}
