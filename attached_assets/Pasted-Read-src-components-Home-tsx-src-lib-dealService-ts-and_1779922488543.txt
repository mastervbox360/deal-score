Add a deserializeIntoSetters function that takes a saved inputs object and a map of setter functions, and calls each setter with the correct value:

typescript
export function deserializeIntoSetters(
  inputs: SerializedInputs,
  setters: Record<string, (value: unknown) => void>
) {
  Object.entries(inputs).forEach(([key, value]) => {
    if (key in setters && value !== undefined && value !== null) {
      setters[key](value)
    }
  })
}
