function switchAnalysisView(view){
  analysisViews.forEach(function(v){
    var el = document.getElementById('view-'+v);
    var pill = document.getElementById('pill-'+v);
    var lb = document.getElementById('lb-'+v);

    inputs:      { icon:'ti-checks',       text:'Confirm inputs',   fn:"window.location.href='ds_analysis_hub.html?view=results'" },

/* ── Next action / sidebar alignment helper ── */
function syncSidebarAlign(){
  var sidebars = document.querySelectorAll('.sbar-sticky');
  if(!sidebars.length) return;
  var expCard = document.querySelector('.exp-card');
  var expVisible = expCard && expCard.style.display !== 'none';
  sidebars.forEach(function(sidebar){
    /* Sensitivity sidebar always sits flush — no banner offset needed */
    if(sidebar.closest('#view-sensitivity') || sidebar.closest('#view-workings')){
      sidebar.style.marginTop = '0';
      return;
    }
    if(expVisible){ sidebar.style.marginTop = '0'; }
    else {
      var pill = document.querySelector('.exp-restore.show');
      var offset = pill ? (pill.offsetHeight + 14) : 10;
      sidebar.style.marginTop = offset + 'px';
    }
  });
}

function scRenderStep(){
  var step = scSteps[scCurrentStep];
  var total = scSteps.length;
  document.getElementById('sc-step-num').textContent = scCurrentStep + 1;
  document.getElementById('sc-step-total').textContent = total;
  document.getElementById('sc-progress-bar').style.width = (((scCurrentStep + 1) / total) * 100) + '%';
  document.getElementById('sc-field-label').textContent = step.label;
  document.getElementById('sc-field-current').textContent = step.current;
  document.getElementById('sc-field-icon').innerHTML = step.icon;
  document.getElementById('sc-tip-text').textContent = step.tip;
  /* input */
  var wrap = document.getElementById('sc-input-wrap');
  wrap.innerHTML = '<input id="sc-field-input" type="' + step.inputType + '" placeholder="' + step.inputPlaceholder + '" style="width:100%;padding:7px 10px;border:.5px solid var(--border);border-radius:7px;font-size:12px;font-family:inherit;color:var(--text-1);background:#fff;outline:none;" />';
  wrap.querySelector('input').addEventListener('keydown', function(e){ if(e.key==='Enter') scSubmitStep(); });
  /* source link */
  var sw = document.getElementById('sc-source-wrap');
  if(step.source){
    sw.style.display = 'block';
    document.getElementById('sc-source-link').href = step.source.url;
    document.getElementById('sc-source-label').textContent = step.source.label;
  } else {
    sw.style.display = 'none';
  }
  /* dots */
  var dots = document.getElementById('sc-dots');
  dots.innerHTML = '';
  for(var i=0;i<total;i++){
    var d = document.createElement('div');
    d.style.cssText = 'width:6px;height:6px;border-radius:50%;background:' + (i < scCurrentStep ? 'var(--teal)' : i === scCurrentStep ? '#fff' : 'rgba(255,255,255,.3)') + ';transition:background .2s;';
    dots.appendChild(d);
  }
  /* badge count */
  var remaining = total - scCompleted.length;
  var badge = document.getElementById('assist-badge');
  if(badge) badge.textContent = remaining;
  document.getElementById('sc-complete').style.display = 'none';
  document.getElementById('sc-step-num').closest('div').parentElement.style.display = '';
}

function scSubmitStep(){
  var input = document.getElementById('sc-field-input');
  var val = input ? input.value.trim() : '';
  if(!val){ input.style.borderColor='#f87171'; setTimeout(function(){ input.style.borderColor=''; },800); return; }
  scCompleted.push({ field: scSteps[scCurrentStep].label, value: val });
  scCurrentStep++;
  if(scCurrentStep >= scSteps.length){
    /* all done */
    document.getElementById('sc-step-num').closest('div').parentElement.style.display = 'none';
    document.getElementById('sc-complete').style.display = 'block';
    var badge = document.getElementById('assist-badge');
    if(badge){ badge.textContent='0'; badge.style.background='var(--teal-light)'; badge.style.color='var(--teal)'; }
    document.getElementById('assist-sub').textContent = 'All fields confirmed — score updated';
    showToast('Smart Capture complete — results recalculated');
  } else {
    scRenderStep();
    showToast('Saved · moving to next field');
  }
}

function scSkipStep(){
  scCurrentStep++;
  if(scCurrentStep >= scSteps.length){ scCurrentStep = 0; }
  scRenderStep();
  showToast('Skipped — you can come back to this later');
}

function notesRelTime(ts){
  var diff = Date.now() - ts;
  var mins = Math.round(diff/60000);
  if(mins < 1) return 'just now';
  if(mins < 60) return mins + (mins===1 ? ' minute ago' : ' minutes ago');
  var hrs = Math.round(mins/60);
  if(hrs < 24) return hrs + (hrs===1 ? ' hour ago' : ' hours ago');
  var days = Math.round(hrs/24);
  return days + (days===1 ? ' day ago' : ' days ago');
}

function updateOptimiser(){
  var baseAsk=192000, baseRent=1150, baseCoc=8.2, baseCF=312, baseBreakeven=838;
  var targetStr = (document.getElementById('opt-target-val')||{}).value||'8.0%';
  var targetNum = parseFloat(targetStr.replace(/[^0-9.]/g,''))||8;
  /* derive max price from CoC target — simplified: price = (rent*12 - costs) / targetCoc * 100 */
  var annualRent = baseRent*12;
  var annualCosts = 3200; /* running costs */
  var annualMortgageCost = 0.054; /* rate */
  var deposit = 0.25;
  /* for CoC: cash invested = price*deposit + 8400 + 3200, net income = rent*12 - costs - mortgage interest */
  var maxPrice, headroom, minRent, rentBuf, targetMet;
  if(optTarget==='coc'){
    /* maxPrice so that (annualRent - annualCosts - maxPrice*(1-deposit)*0.054) / (maxPrice*deposit+11600) = targetNum/100 */
    maxPrice = Math.round((annualRent-annualCosts) / (targetNum/100 * deposit + (1-deposit)*0.054) / 500)*500;
    headroom = maxPrice - baseAsk;
    minRent = Math.round(baseBreakeven + (baseCF * targetNum/8.2));
    rentBuf = baseRent - minRent;
    targetMet = baseCoc >= targetNum;
  } else if(optTarget==='cf'){
    var targetCF = targetNum;
    maxPrice = Math.round((annualRent - annualCosts - 838*12 - targetCF*12) / (annualMortgageCost*(1-deposit)) / 500)*500;
    maxPrice = Math.max(100000, maxPrice);
    headroom = maxPrice - baseAsk;
    minRent = Math.round(838 + targetCF + annualCosts/12);
    rentBuf = baseRent - minRent;
    targetMet = baseCF >= targetNum;
  } else if(optTarget==='yield'){
    maxPrice = Math.round((baseRent*12) / (targetNum/100) / 500)*500;
    headroom = maxPrice - baseAsk;
    minRent = Math.round(baseAsk * targetNum/100 / 12);
    rentBuf = baseRent - minRent;
    targetMet = (baseRent*12/baseAsk*100) >= targetNum;
  } else if(optTarget==='net'){
    maxPrice = Math.round((baseRent*12 - annualCosts) / (targetNum/100) / 500)*500;
    headroom = maxPrice - baseAsk;
    minRent = Math.round((baseAsk * targetNum/100 + annualCosts) / 12);
    rentBuf = baseRent - minRent;
    targetMet = ((baseRent*12-annualCosts)/baseAsk*100) >= targetNum;
  } else { /* max cash in */
    maxPrice = baseAsk;
    headroom = 0;
    minRent = baseBreakeven + 100;
    rentBuf = baseRent - minRent;
    targetMet = (baseAsk*deposit + 11600) <= targetNum*1000;
  }
  headroom = Math.round(headroom/100)*100;
  var el = function(id){ return document.getElementById(id); };
  if(el('opt-max-price')) el('opt-max-price').textContent = '£'+maxPrice.toLocaleString();
  if(el('opt-headroom')){
    el('opt-headroom').textContent = (headroom>=0?'+':'')+'£'+Math.abs(headroom).toLocaleString();
    el('opt-headroom').className = 'opt-rv '+(headroom>=0?'gr':'warn');
  }
  if(el('opt-min-rent')) el('opt-min-rent').textContent = '£'+Math.max(700,minRent)+'/mo';
  if(el('opt-rent-buf')){
    var buf = Math.round(rentBuf);
    el('opt-rent-buf').textContent = buf>=0 ? '£'+buf+'/mo buffer' : '£'+Math.abs(buf)+'/mo shortfall';
    el('opt-rent-buf').className = 'opt-rv '+(buf>=30?'gr':'warn');
  }
  if(el('opt-verdict-label')) el('opt-verdict-label').textContent = optTargetLabels[optTarget]+' target: '+targetStr;
  if(el('opt-verdict-pill')){
    el('opt-verdict-pill').className = 'opt-verdict-pill '+(targetMet?'ovp-hit':'ovp-miss');
    el('opt-verdict-pill').innerHTML = targetMet ? '<i class="ti ti-check" style="font-size:11px;"></i> Target met' : '<i class="ti ti-alert-triangle" style="font-size:11px;"></i> Negotiate needed';
  }
  if(el('opt-negotiation-note')) el('opt-negotiation-note').innerHTML = optNegotiationTips[optTarget]||'';
}

function deletePhoto(idx){
  var wasHero = (heroIdx === idx);

    if(heroIdx >= 0 && photoData[heroIdx]){

    } else {
      try { localStorage.removeItem('dealHeroPhoto_DS-004'); } catch(ex){}

}
